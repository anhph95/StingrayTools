#   pip install ".[sensors]"
#   pip install ".[images]"
#   pip install ".[ctd]"
#   pip install ".[abundance]"
#

#   pip install ".[pipeline]"
#
# For local Stingray CLI development:
#   pip install -e ".[dev]"
#
# Dashboard is a separate package:
#   pip install -e "./packages/stingray-dashboard"
#
# Legacy Tator support is retained but not installed by default:
#   pip install ".[legacy-tator]"
